cmake_minimum_required(VERSION 3.21)
project(NativeOpenSmoke LANGUAGES CXX)
if(NOT APPLE)
    message(FATAL_ERROR "This harness requires a logged-in macOS GUI session")
endif()
get_filename_component(DEFAULT_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(REPO "${DEFAULT_REPO}" CACHE PATH "Production source checkout")
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(native_open_receiver native_open_receiver.cpp)
target_compile_features(native_open_receiver PRIVATE cxx_std_17)
option(USE_TORRENT "Exercise torrent-enabled production queue" ON)
if(USE_TORRENT)
    target_compile_definitions(native_open_receiver PRIVATE USE_TORRENT)
endif()
target_include_directories(native_open_receiver PRIVATE "${REPO}/eiskaltdcpp-qt/src")
target_link_libraries(native_open_receiver PRIVATE Qt6::Widgets)
