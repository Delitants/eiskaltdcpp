cmake_minimum_required(VERSION 3.21)
project(CocoaAccessibilityRegression LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "Requires a logged-in macOS GUI session")
endif()
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Widgets)
add_executable(cocoa_accessibility_regression main.mm)
target_compile_features(cocoa_accessibility_regression PRIVATE cxx_std_17)
target_link_libraries(cocoa_accessibility_regression PRIVATE Qt6::Widgets "-framework AppKit")
